36void nuskaitytiFaila(Konteineris& studentai, const std::string& failoVardas)
37{
38 std::ifstream failas(failoVardas);
39 if (!failas) {
40 throw std::runtime_error("Nepavyko atidaryti failo " + failoVardas);
41 }
42
43 std::stringstream buffer;
44 buffer << failas.rdbuf();
45
46 std::string eilute;
47 std::getline(buffer, eilute);
48
49 while (std::getline(buffer, eilute)) {
50 std::stringstream ss(eilute);
51
52 Studentas s;
53 std::string vardas, pavarde;
54 if (!(ss >> vardas >> pavarde)) {
55 throw std::runtime_error("Blogas formatas eiluteje " + eilute);
56 }
57 s.setVardas(vardas);
58 s.setPavarde(pavarde);
59
60 int pazymys;
61 while (ss >> pazymys) {
62 s.pridetiNd(pazymys);
63 }
64
65 if (s.getNd().empty()) {
66 throw std::runtime_error("Truksta pazymiu eiluteje " + eilute);
67 }
68
69 s.setEgz(s.getNd().back());
70 s.pasalintiPaskutiniNd();
71
72 s.setGalVid(s.skaiciuotiGalutiniSuVid());
73 s.setGalMed(s.skaiciuotiGalutiniSuMed());
74
75 studentai.push_back(s);
76 }
77}
